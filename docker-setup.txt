# Docker Hub login:
Use login through github

# Pull an image
docker pull [namespace]/[name]
ex: docker pull docker/getting-started

- View images:
docker images

# Run a container:
docker run -d -p hostport:containerport namespace/name:tage

# View running containers:
docker ps

# Stop/kill containers;
docker stop CONTAINER_ID -> uses SIGTERM to stop the container **IDEAL**
docker kill CONTAINER_ID -> uses SIGKILL to kill the container

# Create volume
Volumes are file systems that live outside a container that the container has access to
docker volume create VOLUME_NAME -> create volume VOLUME_NAME
docker volume ls -> lists volumes
docker volume inspect VOLUME_NAME -> see where VOLUME_NAME is on your local machine


# Remove container
docker rm CONTAINER_ID

# Run scripts from inside a container:
docker exec CONTAINER_ID ls -> list container contents
docker exec -it CONTAINERID /bin/sh -> get a terminal from inside the container

# Track resource usage:
docker stats [CONTAINER_ID] -> live feedback of resource usage by container/containers
docker top CONTAINER -> shows processes running inside the container
